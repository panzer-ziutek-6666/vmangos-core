cmake_minimum_required(VERSION 3.24)
project(VmangosAccountsBootstrapA142 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "A142 requires Windows MSVC x64")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(Vmangos.Accounts.Bootstrap bootstrap_win32.cpp)
target_compile_definitions(Vmangos.Accounts.Bootstrap PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
target_link_libraries(Vmangos.Accounts.Bootstrap PRIVATE advapi32 bcrypt)
add_executable(Vmangos.Accounts.Bootstrap.Tests bootstrap_tests.cpp)
foreach(target Vmangos.Accounts.Bootstrap Vmangos.Accounts.Bootstrap.Tests)
  target_compile_options(${target} PRIVATE /W4 /WX /permissive- /EHsc /utf-8 /guard:cf /sdl)
  target_link_options(${target} PRIVATE /DYNAMICBASE /NXCOMPAT /HIGHENTROPYVA /guard:cf /INCREMENTAL:NO)
endforeach()
